<script>

const questions = {

  A: [
    "Usaha memiliki akses internet yang memadai untuk kegiatan usaha.",
    "Usaha menggunakan pembayaran digital seperti QRIS.",
    "Usaha menggunakan media sosial untuk kegiatan pemasaran.",
    "Usaha menggunakan kanal digital untuk menerima pesanan atau transaksi."
  ],

  C: [
    "Usaha memiliki katalog produk digital yang diperbarui secara berkala.",
    "Usaha menggunakan data transaksi untuk memahami penjualan.",
    "Usaha mengelola informasi atau hubungan dengan pelanggan secara digital.",
    "Usaha memiliki praktik keamanan akun dan data digital yang memadai."
  ],

  M: [
    "Usaha memiliki lebih dari satu kanal penjualan.",
    "Usaha memiliki pelanggan dari luar wilayah utama usaha.",
    "Usaha mampu menggunakan kanal digital alternatif ketika satu kanal terganggu.",
    "Usaha secara aktif mencari atau mengembangkan pelanggan baru melalui kanal digital."
  ],

  T: [
    "Usaha mampu menyesuaikan produk ketika terjadi perubahan kebutuhan pelanggan.",
    "Usaha mampu mengubah metode pemesanan atau penjualan ketika diperlukan.",
    "Usaha memiliki alternatif pemasok atau cara memperoleh bahan/input usaha.",
    "Usaha mampu menyesuaikan proses operasi ketika terjadi gangguan."
  ]

};


function createQuestions() {

  Object.keys(questions).forEach(dim => {

    const container =
      document.getElementById("questions" + dim);

    questions[dim].forEach((q, index) => {

      const name = dim + index;

      container.innerHTML += `

        <div class="question">

          <div class="question-title">
            ${index + 1}. ${q}
          </div>

          <div class="scale">

            <label>
              <input type="radio" name="${name}" value="1">
              <span>1<br>Sangat rendah</span>
            </label>

            <label>
              <input type="radio" name="${name}" value="2">
              <span>2<br>Rendah</span>
            </label>

            <label>
              <input type="radio" name="${name}" value="3">
              <span>3<br>Sedang</span>
            </label>

            <label>
              <input type="radio" name="${name}" value="4">
              <span>4<br>Tinggi</span>
            </label>

            <label>
              <input type="radio" name="${name}" value="5">
              <span>5<br>Sangat tinggi</span>
            </label>

          </div>

        </div>

      `;

    });

  });

}


function getDimensionScore(dim) {

  let total = 0;

  questions[dim].forEach((q, index) => {

    const selected =
      document.querySelector(
        `input[name="${dim + index}"]:checked`
      );

    if (!selected) {
      throw new Error(
        "Mohon lengkapi seluruh pertanyaan."
      );
    }

    total += Number(selected.value);

  });

  const max =
    questions[dim].length * 5;

  return Math.round((total / max) * 100);

}


function category(score) {

  if (score < 40) {
    return "Rendah";
  }

  if (score < 60) {
    return "Rentan";
  }

  if (score < 80) {
    return "Berkembang";
  }

  return "Tinggi";

}


function calculateScore() {

  try {

    const A = getDimensionScore("A");
    const C = getDimensionScore("C");
    const M = getDimensionScore("M");
    const T = getDimensionScore("T");


    /*
      Rumus sesuai kerangka penelitian:

      DRI = (A + C + M + T) / 4
    */

    const DRI =
      Math.round((A + C + M + T) / 4);


    document.getElementById("totalScore")
      .innerText = DRI;

    document.getElementById("category")
      .innerText = category(DRI);


    showDimension("A", A);
    showDimension("C", C);
    showDimension("M", M);
    showDimension("T", T);


    /*
      Untuk segmentasi prototype:

      Capability = C

      Resilience = rata-rata A, M, T

      Ini merupakan operationalisasi prototype,
      bukan hasil validasi empiris.
    */

    const resilience =
      Math.round((A + M + T) / 3);


    generateSegment(C, resilience);

    generateRecommendations(A, C, M, T);


    document.getElementById("result")
      .style.display = "block";


    document.getElementById("result")
      .scrollIntoView({
        behavior: "smooth"
      });

  }

  catch(error) {

    alert(error.message);

  }

}


function showDimension(dim, score) {

  document.getElementById(
    "score" + dim
  ).innerText = score;

  document.getElementById(
    "bar" + dim
  ).style.width = score + "%";

}


function generateSegment(capability, resilience) {

  let name = "";
  let description = "";


  if (capability < 60 && resilience < 60) {

    name = "Digital Fragile";

    description =
      "Kapabilitas digital dan ketahanan relatif rendah. " +
      "Prioritas diarahkan pada penguatan fondasi digital, " +
      "literasi, keamanan digital, pencatatan sederhana, " +
      "dan penggunaan pembayaran digital.";

  }


  else if (capability >= 60 && resilience < 60) {

    name = "Digitally Active but Vulnerable";

    description =
      "Kapabilitas digital relatif lebih baik, tetapi " +
      "ketahanan masih rendah. Fokus diarahkan pada " +
      "diversifikasi pasar, manajemen stok, pemasok, " +
      "dan kesiapan menghadapi gangguan.";

  }


  else if (capability < 60 && resilience >= 60) {

    name = "Resilient but Digitally Shallow";

    description =
      "Ketahanan relatif baik tetapi kapabilitas digital " +
      "masih terbatas. Digitalisasi dilakukan secara " +
      "bertahap sesuai kebutuhan usaha, bukan melalui " +
      "pelatihan digital yang seragam.";

  }


  else {

    name = "Digital Resilient";

    description =
      "Kapabilitas digital dan ketahanan relatif tinggi. " +
      "UMKM dapat diarahkan pada scale-up, business matching, " +
      "onboarding pemasok, dan perluasan pasar antardaerah.";

  }


  document.getElementById(
    "segmentName"
  ).innerText = name;


  document.getElementById(
    "segmentDescription"
  ).innerText = description;

}


function generateRecommendations(A, C, M, T) {

  const recommendations = [];


  if (A < 60) {

    recommendations.push(
      "Perkuat akses dan penggunaan dasar digital, " +
      "termasuk pembayaran digital dan kanal digital."
    );

  }


  if (C < 60) {

    recommendations.push(
      "Prioritaskan literasi dan kapabilitas digital, " +
      "pencatatan transaksi, pengelolaan pelanggan, " +
      "katalog digital, serta keamanan akun."
    );

  }


  if (M < 60) {

    recommendations.push(
      "Prioritaskan diversifikasi kanal penjualan, " +
      "pengembangan pelanggan baru, dan perluasan pasar."
    );

  }


  if (T < 60) {

    recommendations.push(
      "Perkuat kemampuan adaptasi terhadap perubahan " +
      "produk, pemasok, metode pemesanan, dan operasi."
    );

  }


  if (recommendations.length === 0) {

    recommendations.push(
      "Pertahankan kapabilitas dan ketahanan yang telah baik. " +
      "Prioritas berikutnya dapat diarahkan pada scale-up, " +
      "business matching, onboarding pemasok, dan perluasan pasar."
    );

  }


  document.getElementById(
    "recommendations"
  ).innerHTML =
    recommendations
      .map(
        item =>
          `<div class="recommendation">✓ ${item}</div>`
      )
      .join("");

}


createQuestions();
